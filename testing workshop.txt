-- =====================================================================
-- CONFIGURATION
-- Tentukan ID workshop yang akan digunakan untuk testing
-- =====================================================================
SET @workshop_id = 1;



-- =====================================================================
-- TEST 0: CEK KONDISI AWAL WORKSHOP
-- Digunakan untuk melihat kondisi sebelum melakukan perubahan
-- =====================================================================
SELECT
    id,
    workshop_name,
    is_active,
    subscription_status,
    subscription_end,
    FROM_UNIXTIME(subscription_end / 1000) AS subscription_end_date
FROM workshops
WHERE id = @workshop_id;



-- =====================================================================
-- TEST 0.1: CEK RIWAYAT PEMBAYARAN LANGGANAN
-- Melihat seluruh transaksi subscription milik workshop
-- =====================================================================
SELECT
    id,
    workshop_id,
    order_id,
    amount,
    transaction_id,
    payment_type,
    transaction_status,
    FROM_UNIXTIME(paid_at / 1000) AS paid_date,
    is_delete,
    FROM_UNIXTIME(created_at / 1000) AS created_date
FROM subscription_payments
WHERE workshop_id = @workshop_id
ORDER BY created_at DESC;



-- =====================================================================
-- TEST 1: BERSIHKAN TRANSAKSI PENDING
-- Digunakan apabila transaksi pending lama menghalangi pembuatan
-- pembayaran subscription baru
--
-- Data tidak dihapus permanen, tetapi dilakukan soft delete
-- =====================================================================
UPDATE subscription_payments
SET
    transaction_status = 'expire',
    is_delete = 1,
    deleted_at = UNIX_TIMESTAMP(NOW(3)) * 1000,
    updated_at = UNIX_TIMESTAMP(NOW(3)) * 1000
WHERE workshop_id = @workshop_id
  AND transaction_status = 'pending'
  AND is_delete = 0;



-- =====================================================================
-- TEST 2: KONDISI BELUM BERLANGGANAN
--
-- Expected:
-- - Banner: Belum Berlangganan
-- - subscription_status = 0
-- - subscription_end = NULL
-- - CRUD dan transaksi menghasilkan HTTP 402
-- - Halaman read, dashboard, riwayat, dan laporan tetap bisa dibuka
-- =====================================================================
UPDATE workshops
SET
    is_active = 1,
    subscription_status = 0,
    subscription_end = NULL,
    updated_at = UNIX_TIMESTAMP(NOW(3)) * 1000
WHERE id = @workshop_id;

-- =====================================================================
-- TEST 2.1: SIMULASI WORKSHOP BENAR-BENAR BELUM PERNAH BERLANGGANAN
--
-- Blok ini opsional.
-- Semua riwayat pembayaran aktif akan disembunyikan dengan soft delete.
-- Gunakan hanya ketika ingin menguji tampilan "belum pernah berlangganan".
$# Catatan penting: blok Test 2.1 melakukan soft delete seluruh riwayat 
   pembayaran aktif. Jangan jalankan blok itu lagi apabila riwayat pembayaran hasil testing Midtrans masih ingin ditampilkan.
-- =====================================================================
UPDATE subscription_payments
SET
    is_delete = 1,
    deleted_at = UNIX_TIMESTAMP(NOW(3)) * 1000,
    updated_at = UNIX_TIMESTAMP(NOW(3)) * 1000
WHERE workshop_id = @workshop_id
  AND is_delete = 0;



-- =====================================================================
-- TEST 3: LANGGANAN AKTIF 30 HARI
--
-- Expected:
-- - Banner subscription tidak tampil
-- - Status langganan aktif
-- - CRUD dan transaksi dapat digunakan
-- - Tombol perpanjangan belum aktif karena sisa lebih dari 7 hari
-- =====================================================================
UPDATE workshops
SET
    is_active = 1,
    subscription_status = 1,
    subscription_end =
        UNIX_TIMESTAMP(
            DATE_ADD(NOW(), INTERVAL 30 DAY)
        ) * 1000,
    updated_at = UNIX_TIMESTAMP(NOW(3)) * 1000
WHERE id = @workshop_id;
